Data Access Patterns

This document describes how the Riello UPS Servers Shutdown system accesses and manipulates data in the SQLite database.

Connection Management

The system uses short-lived database connections:

# Pattern: Open connection, execute, close immediately
def save_client(self, hostname: str, ip_address: str, port: int):
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute('''
        INSERT OR REPLACE INTO client_connections 
        (hostname, ip_address, port, last_connection_time)
        VALUES (?, ?, ?, ?)
    ''', (hostname, ip_address, port, datetime.now().isoformat()))
    
    conn.commit()
    conn.close()

Database Path Resolution

# Server determines path relative to script location
self.db_path = os.path.join(os.path.dirname(__file__), 'ups_clients.db')

# Dashboard uses same location
DB_PATH = os.path.join(os.path.dirname(__file__), 'ups_clients.db')

Client Operations

Save New Client

def save_client_to_db(self, hostname: str, ip_address: str, port: int):
    """Save or update client in database."""
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    # Get existing shutdown delay if client exists
    cursor.execute(
        'SELECT seconds_to_shutdown FROM client_connections WHERE hostname = ?',
        (hostname,)
    )
    row = cursor.fetchone()
    shutdown_delay = row[0] if row else 0
    
    # Insert or update client
    cursor.execute('''
        INSERT OR REPLACE INTO client_connections 
        (hostname, ip_address, port, last_connection_time, seconds_to_shutdown)
        VALUES (?, ?, ?, ?, ?)
    ''', (hostname, ip_address, port, datetime.now().isoformat(), shutdown_delay))
    
    conn.commit()
    conn.close()

Update Heartbeat Timestamp

def update_client_heartbeat(self, hostname: str):
    """Update the last connection time for a client."""
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute('''
        UPDATE client_connections 
        SET last_connection_time = ?
        WHERE hostname = ?
    ''', (datetime.now().isoformat(), hostname))
    
    conn.commit()
    conn.close()

Get All Clients

def get_all_clients(self) -> list:
    """Retrieve all client records from database."""
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute('''
        SELECT hostname, ip_address, port, last_connection_time, seconds_to_shutdown
        FROM client_connections
        ORDER BY hostname
    ''')
    
    clients = cursor.fetchall()
    conn.close()
    
    return clients

Update Shutdown Delay

def update_shutdown_delay(self, hostname: str, seconds: int) -> bool:
    """Update shutdown delay for a specific client.
    
    Returns:
        bool: True if client was found and updated, False otherwise
    """
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute('''
        UPDATE client_connections 
        SET seconds_to_shutdown = ?
        WHERE hostname = ?
    ''', (seconds, hostname))
    
    rows_affected = cursor.rowcount
    conn.commit()
    conn.close()
    
    return rows_affected > 0

Remove Client

def remove_client_from_db(self, hostname: str):
    """Remove a client from the database."""
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute(
        'DELETE FROM client_connections WHERE hostname = ?',
        (hostname,)
    )
    
    conn.commit()
    conn.close()

Configuration Operations

Get Configuration Value

def get_config(self, key: str, default: str = None) -> str:
    """Get a configuration value by key.
    
    Args:
        key: Configuration key name
        default: Default value if key not found
    
    Returns:
        str: Configuration value or default
    """
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute(
        'SELECT value FROM configuration WHERE key = ?',
        (key,)
    )
    
    row = cursor.fetchone()
    conn.close()
    
    return row[0] if row else default

Set Configuration Value

def set_config(self, key: str, value: str):
    """Set a configuration value.
    
    Uses INSERT OR REPLACE to handle both new and existing keys.
    """
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute('''
        INSERT OR REPLACE INTO configuration (key, value)
        VALUES (?, ?)
    ''', (key, value))
    
    conn.commit()
    conn.close()

Get All Configuration

def get_all_config(self) -> dict:
    """Get all configuration as a dictionary."""
    conn = sqlite3.connect(self.db_path)
    cursor = conn.cursor()
    
    cursor.execute('SELECT key, value FROM configuration')
    
    config = {row[0]: row[1] for row in cursor.fetchall()}
    conn.close()
    
    return config

Dashboard Data Access

The dashboard uses similar patterns but returns JSON-serializable data:

API: Get Clients

def handle_api_clients(self):
    """Return all clients as JSON."""
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    
    cursor.execute('''
        SELECT hostname, ip_address, port, last_connection_time, seconds_to_shutdown
        FROM client_connections
        ORDER BY hostname
    ''')
    
    clients = []
    for row in cursor.fetchall():
        clients.append({
            'hostname': row[0],
            'ip_address': row[1],
            'port': row[2],
            'last_connection_time': row[3],
            'seconds_to_shutdown': row[4]
        })
    
    conn.close()
    
    return json.dumps({'clients': clients})

API: Update Configuration

def handle_update_config(self, params: dict) -> str:
    """Update configuration from dashboard form.
    
    Args:
        params: Dictionary with 'ups_url' and 'minimum_minutes'
    
    Returns:
        JSON response string
    """
    ups_url = params.get('ups_url', [''])[0]
    minimum_minutes = params.get('minimum_minutes', ['15'])[0]
    
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    
    cursor.execute(
        'INSERT OR REPLACE INTO configuration (key, value) VALUES (?, ?)',
        ('UPS_URL', ups_url)
    )
    cursor.execute(
        'INSERT OR REPLACE INTO configuration (key, value) VALUES (?, ?)',
        ('UPS_minimum_minutes', minimum_minutes)
    )
    
    conn.commit()
    conn.close()
    
    return json.dumps({'success': True, 'message': 'Configuration updated'})

Error Handling

Database operations should handle SQLite exceptions:

def safe_db_operation(self):
    """Example of safe database operation with error handling."""
    try:
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('SELECT * FROM client_connections')
        result = cursor.fetchall()
        
        conn.close()
        return result
        
    except sqlite3.Error as e:
        logging.error(f"Database error: {e}")
        return None
    except Exception as e:
        logging.error(f"Unexpected error accessing database: {e}")
        return None

Thread Safety

SQLite in Python is thread-safe when:

  • Each thread uses its own connection
  • Connections are not shared between threads

The current implementation opens a new connection for each operation, which is thread-safe but has some overhead.

# Thread-safe pattern used in the system
def operation_in_any_thread(self):
    # Each call creates its own connection
    conn = sqlite3.connect(self.db_path)
    try:
        # ... do work ...
        conn.commit()
    finally:
        conn.close()  # Always close

Data Migration

If schema changes are needed, manual migration is required:

#!/bin/bash
# Example migration script

# Backup existing database
cp /opt/UPSserver/ups_clients.db /opt/UPSserver/ups_clients.db.backup

# Apply migration
sqlite3 /opt/UPSserver/ups_clients.db <<EOF
-- Add new column example
ALTER TABLE client_connections ADD COLUMN new_field TEXT DEFAULT '';
EOF

echo "Migration complete"
