Domain Entities

This document describes the core domain entities in the Riello UPS Servers Shutdown system.

ClientConnection

Location: server/UPSserver.py

Purpose: Represents a connected client machine managed by the UPS server.

Properties

PropertyTypeDescription
hostnamestrUnique identifier - the client's hostname
ip_addressstrClient's IP address
portintConnection port number
connectionsocketActive TCP socket connection
last_heartbeatfloatUnix timestamp of last heartbeat received
seconds_to_shutdownintDelay before shutdown (default: 0)

Code Definition

class ClientConnection:
    """Represents a connected client with its socket and metadata."""
    
    def __init__(self, hostname: str, ip_address: str, port: int, connection: socket.socket):
        self.hostname = hostname
        self.ip_address = ip_address
        self.port = port
        self.connection = connection
        self.last_heartbeat = time.time()
        self.seconds_to_shutdown = 0  # Default shutdown delay

Usage Example

# Creating a new client connection
client = ClientConnection(
    hostname="web-server-01",
    ip_address="192.168.1.100",
    port=5226,
    connection=client_socket
)

# Updating heartbeat
client.last_heartbeat = time.time()

# Setting shutdown delay
client.seconds_to_shutdown = 30  # 30 second delay

Code Definition

class ReadUPSMinutes:
    """Reads the remaining battery minutes from the UPS JSON API."""
    
    def __init__(self, url: str):
        self.url = url
    
    def get_minutes(self) -> int:
        """Fetch the autonomy value from the UPS API.
        
        Returns:
            int: Remaining battery minutes, or None if request fails
        """
        ctx = ssl.create_default_context()
        ctx.check_hostname = False
        ctx.verify_mode = ssl.CERT_NONE
        
        try:
            req = urllib.request.Request(self.url)
            with urllib.request.urlopen(req, timeout=10, context=ctx) as response:
                data = json.loads(response.read().decode())
                return data.get('autonomy')
        except Exception as e:
            logging.error(f"Error reading UPS data: {e}")
            return None
